    <script>
        const inputEmployees = document.getElementById('employees');
        const inputHours = document.getElementById('hours');
        const inputWage = document.getElementById('wage');
        const inputEfficiency = document.getElementById('efficiency');

        const displayCurrentCost = document.getElementById('current-cost');
        const displaySavings = document.getElementById('savings-value');
        
        const barWaste = document.getElementById('bar-waste');
        const barSavings = document.getElementById('bar-savings');

        function calculateROI() {
            const employees = parseFloat(inputEmployees.value) || 0;
            const hoursPerDay = parseFloat(inputHours.value) || 0;
            const monthlySalary = parseFloat(inputWage.value) || 0;
            const efficiencyGain = (parseFloat(inputEfficiency.value) || 0) / 100;

            // 160 standard working hours in a month
            const standardMonthlyHours = 160;
            const hourlyWage = monthlySalary / standardMonthlyHours;
            const totalManualHoursPerMonth = employees * hoursPerDay * 20;

            const currentMonthlyCost = totalManualHoursPerMonth * hourlyWage;
            const monthlySavings = currentMonthlyCost * efficiencyGain;

            // Format numbers to Ghanaian New Cedi currency layout
            displayCurrentCost.innerText = `GHS ${currentMonthlyCost.toLocaleString('en-GH', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
            displaySavings.innerText = `GHS ${monthlySavings.toLocaleString('en-GH', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;

            // Dynamically resize bars with basic math calculations instead of a heavy library
            if (currentMonthlyCost > 0) {
                barWaste.style.width = "100%";
                barSavings.style.width = `${efficiencyGain * 100}%`;
            } else {
                barWaste.style.width = "0%";
                barSavings.style.width = "0%";
            }
        }

        // Attach clean event tracking listeners
        [inputEmployees, inputHours, inputWage, inputEfficiency].forEach(input => {
            input.addEventListener('input', calculateROI);
        });

        // Run calculation instantly on page load
        calculateROI();
    </script>
